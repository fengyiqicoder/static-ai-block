<script type="module">
  const form = document.getElementById("ask-form"), input = document.getElementById("ask-input"), thread = document.getElementById("ask-thread");
  const root = document.querySelector("main") || document.body;
  // The page is sent as numbered passages, so the answer can cite them as [1], [2]…
  const BLOCKS = "h1,h2,h3,h4,p,li,td,pre,blockquote";
  const passages = () => [...root.querySelectorAll(BLOCKS)].filter((el) => el.innerText.trim() && !el.querySelector(BLOCKS) && !el.closest("#sab-ask"));
  const esc = (s) => s.replace(/[&<>]/g, (c) => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;" }[c]));
  const history = [];                                   // earlier turns, so follow-ups work
  let sources = [];

  // [2] or [2, 5] in the answer becomes a button for each passage it names.
  const render = (text) => esc(text).replace(/\[(\d+(?:\s*,\s*\d+)*)\]/g, (m, list) =>
    list.split(",").map((n) => sources[n.trim() - 1] ? `<button class="ask-cite" type="button" data-n="${n.trim()}" title="Show the source">${n.trim()}</button>` : `[${n}]`).join(""));

  async function ask(q) {
    sources = passages();
    const turn = document.createElement("div"); turn.className = "ask-turn";
    turn.innerHTML = `<p class="ask-q">${esc(q)}</p><div class="sab-out sab-wait"></div><p class="ask-src" hidden></p>`;
    thread.append(turn);
    const out = turn.querySelector(".sab-out");
    let text = "";
    try {
      const page = sources.length ? sources.map((el, i) => `[${i + 1}] ${el.innerText.trim().replace(/\s+/g, " ")}`).join("\n") : root;
      await onetag.ask(q, page, {
        system: "Answer only from the numbered passages of this page. After each fact, cite the passage it comes from like [3]. If the page does not cover it, say so in one sentence.",
        history, maxTokens: 250,
        stream: (d) => { text += d; out.innerHTML = render(text); },
      });
      history.push({ role: "user", content: q }, { role: "assistant", content: text });
      input.placeholder = "Ask a follow-up…";
    } catch (err) { out.textContent = err.message; }
  }

  // A citation shows the passage under the answer and, when it is on screen, scrolls to it and outlines it.
  thread.addEventListener("click", (e) => {
    const n = e.target.dataset.n, el = sources[n - 1]; if (!el) return;
    const src = e.target.closest(".ask-turn").querySelector(".ask-src");
    src.textContent = `[${n}] ${el.innerText.trim().slice(0, 280)}`; src.hidden = false;
    if (el.getBoundingClientRect().right > 0) {
      el.scrollIntoView({ behavior: "smooth", block: "center" });
      el.classList.add("ask-hit"); setTimeout(() => el.classList.remove("ask-hit"), 2500);
    }
  });
  form.addEventListener("submit", (e) => { e.preventDefault(); const q = input.value.trim(); if (q) { input.value = ""; ask(q); } });
  for (const c of document.querySelectorAll("#sab-ask .sab-chip")) c.addEventListener("click", () => ask(c.dataset.q));
</script>
