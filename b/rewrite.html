<script type="module">
  const box = document.getElementById("rw-text"), out = document.getElementById("rw-diff"), review = document.getElementById("rw-review");
  const esc = (s) => s.replace(/[&<>]/g, (c) => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;" }[c]));
  let result = "";

  // Word-level diff (longest common subsequence): removed words in <del>, added words in <ins>.
  function diff(before, after) {
    const a = before.split(/(\s+)/).filter(Boolean), b = after.split(/(\s+)/).filter(Boolean);
    const t = Array.from({ length: a.length + 1 }, () => new Uint16Array(b.length + 1));
    for (let i = a.length - 1; i >= 0; i--) for (let j = b.length - 1; j >= 0; j--) t[i][j] = a[i] === b[j] ? t[i + 1][j + 1] + 1 : Math.max(t[i + 1][j], t[i][j + 1]);
    let i = 0, j = 0, html = "";
    while (i < a.length || j < b.length) {
      if (i < a.length && j < b.length && a[i] === b[j]) { html += esc(a[i]); i++; j++; }
      else if (i < a.length && (j === b.length || t[i + 1][j] >= t[i][j + 1])) html += `<del>${esc(a[i++])}</del>`;
      else html += `<ins>${esc(b[j++])}</ins>`;
    }
    return html;
  }

  for (const btn of document.querySelectorAll("#sab-rw [data-how]")) {
    btn.addEventListener("click", async () => {
      const original = box.value;
      out.textContent = ""; out.classList.add("sab-wait"); review.hidden = true;
      try {
        result = (await onetag.ask(`${btn.dataset.how} Output only the new text.`, original, {
          stream: (d) => { out.textContent += d; }, maxTokens: 300,
        })).trim();
        out.innerHTML = diff(original, result);          // the stream showed the new text; now show what changed
        review.hidden = false;
      } catch (err) { out.textContent = err.message; }
    });
  }
  const close = () => { out.textContent = ""; out.classList.remove("sab-wait"); review.hidden = true; };
  document.getElementById("rw-accept").addEventListener("click", () => { box.value = result; close(); });
  document.getElementById("rw-discard").addEventListener("click", close);
  document.getElementById("rw-copy").addEventListener("click", async (e) => { await navigator.clipboard.writeText(result); e.target.textContent = "Copied"; setTimeout(() => { e.target.textContent = "Copy"; }, 1500); });
</script>
