<script type="module">
  const box = document.getElementById("rw-text");
  for (const btn of document.querySelectorAll("#rw-btns button")) {
    btn.addEventListener("click", async () => {
      const original = box.value; box.value = "";
      try {
        await onetag.ask(`Rewrite the text to be ${btn.dataset.how}. Keep the meaning. Output only the rewritten text.`, original, {
          stream: (d) => { box.value += d; },
          maxTokens: 300,
        });
      } catch (err) { box.value = original; alert(err.message); }
    });
  }
</script>
