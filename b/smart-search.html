<script type="module">
  const q = document.getElementById("ss-q"), items = [...document.querySelectorAll("#ss-items li")];
  let timer;
  q.addEventListener("input", () => {
    clearTimeout(timer);
    timer = setTimeout(async () => {
      const text = q.value.trim(); items.forEach((li) => li.classList.remove("hit"));
      if (!text) return;
      try {
        const data = await onetag.ask(`Which items match: "${text}"? Return {"ids": [...]} using the ids from the list; empty list if none.`,
          items.map((li) => ({ id: li.dataset.id, text: li.textContent })),   // a plain object is sent as JSON
          { json: true, maxTokens: 60 });
        for (const li of items) if (data.ids.map(String).includes(li.dataset.id)) li.classList.add("hit");
      } catch (err) { console.warn(err.message); }
    }, 500);                                                   // wait for typing to pause
  });
</script>
