<script src="https://onetagai.com/v1.js" data-key="YOUR_ONETAG_KEY"></script>

<div style="max-width:560px">
  <input id="ss-q" type="text" placeholder="e.g. something warm for under 50" style="width:100%;padding:10px 12px;border:1px solid #cbd5e1;border-radius:8px;font:inherit;box-sizing:border-box">
  <ul id="ss-items" style="list-style:none;padding:0;margin:12px 0 0;line-height:1.7">
    <li data-id="1">Merino beanie, grey, $38</li>
    <li data-id="2">Linen summer shirt, white, $64</li>
    <li data-id="3">Fleece-lined gloves, black, $45</li>
    <li data-id="4">Canvas tote bag, natural, $29</li>
    <li data-id="5">Wool scarf, navy, $52</li>
  </ul>
</div>
<style>#ss-items li{padding:4px 8px;border-radius:6px}#ss-items li.hit{background:#fef08a}</style>

<script type="module">
  const q = document.getElementById("ss-q"), items = [...document.querySelectorAll("#ss-items li")];
  let timer;
  q.addEventListener("input", () => {
    clearTimeout(timer);
    timer = setTimeout(async () => {
      const text = q.value.trim(); items.forEach((li) => li.classList.remove("hit"));
      if (!text) return;
      try {
        const data = await onetag.ask(
          `Which items match: "${text}"? Return {"ids": [...]} using the ids from the list; empty list if none.`,
          items.map((li) => ({ id: li.dataset.id, text: li.textContent })),   // a plain object is sent as JSON
          { json: true, maxTokens: 60 }
        );
        for (const li of items) if (data.ids.map(String).includes(li.dataset.id)) li.classList.add("hit");
      } catch (err) { console.warn(err.message); }
    }, 500);                                                   // wait for typing to pause
  });
</script>
