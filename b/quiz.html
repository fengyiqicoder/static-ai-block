<script src="https://onetagai.com/v1.js" data-key="YOUR_ONETAG_KEY"></script>

<div style="max-width:560px">
  <button id="quiz-btn" style="padding:8px 14px;border:0;border-radius:8px;background:#111;color:#fff;font:inherit;cursor:pointer">Quiz me on this page</button>
  <ol id="quiz-out" style="line-height:1.6;padding-left:20px"></ol>
</div>

<script type="module">
  const btn = document.getElementById("quiz-btn"), out = document.getElementById("quiz-out");
  btn.addEventListener("click", async () => {
    out.innerHTML = "<li>…</li>";
    try {
      const data = await onetag.ask(
        'Write 3 multiple-choice questions about the content as JSON: {"questions": [{"q": string, "options": [4 strings], "answer": 0-3}]}',
        document.querySelector("main") || document.body,
        { json: true, maxTokens: 600 }
      );
      out.innerHTML = "";
      for (const item of data.questions) {
        const li = document.createElement("li");
        li.innerHTML = `<b>${item.q}</b><br>` + item.options.map((o, i) => `<label style="display:block"><input type="radio" name="q${out.children.length}" value="${i}"> ${o}</label>`).join("");
        li.addEventListener("change", (e) => { li.style.color = Number(e.target.value) === item.answer ? "#15803d" : "#b91c1c"; });
        out.append(li);
      }
    } catch (err) { out.innerHTML = `<li>${err.message}</li>`; }
  });
</script>
