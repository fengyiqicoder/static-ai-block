<script type="module">
  const play = document.getElementById("ra-play"), stop = document.getElementById("ra-stop"), sum = document.getElementById("ra-sum");
  const rate = document.getElementById("ra-rate"), now = document.getElementById("ra-now"), note = document.getElementById("ra-note");
  const root = document.querySelector("article") || document.querySelector("main") || document.body;
  const voice = window.speechSynthesis;
  let run = 0, state = "idle", marked = null;            // run changes on every start and stop, so old callbacks do nothing

  function mark(el) {                                    // outline the part being read
    if (marked) marked.classList.remove("ra-on");
    marked = el;
    if (el) { el.classList.add("ra-on"); if (el.getBoundingClientRect().right > 0) el.scrollIntoView({ block: "nearest", behavior: "smooth" }); }
  }
  function end() { run++; voice.cancel(); state = "idle"; mark(null); now.textContent = ""; now.classList.remove("sab-wait"); play.textContent = "Listen"; stop.hidden = true; }
  function speak(parts) {                                // parts: [{ text, el }]. One utterance per part: browsers cut off long ones.
    end();
    const id = run; let i = 0;
    state = "playing"; play.textContent = "Pause"; stop.hidden = false;
    const next = () => {
      if (id !== run) return;
      if (i >= parts.length) return end();
      const { text, el } = parts[i++], u = new SpeechSynthesisUtterance(text);
      u.rate = +rate.value; if (document.documentElement.lang) u.lang = document.documentElement.lang;
      u.onend = u.onerror = next;
      now.textContent = text; mark(el); voice.speak(u);
    };
    next();
  }

  if (!voice) { play.disabled = sum.disabled = true; note.textContent = "This browser cannot read text aloud."; }
  play.addEventListener("click", () => {
    if (state === "playing") { voice.pause(); state = "paused"; play.textContent = "Resume"; }
    else if (state === "paused") { voice.resume(); state = "playing"; play.textContent = "Pause"; }
    else speak([...root.querySelectorAll("h1,h2,h3,h4,p,li,blockquote")].filter((el) => el.innerText.trim() && !el.querySelector("p,li") && !el.closest("#sab-read")).map((el) => ({ text: el.innerText.trim(), el })));
  });
  stop.addEventListener("click", end);
  sum.addEventListener("click", async () => {
    end(); now.classList.add("sab-wait"); sum.disabled = true;
    try {
      const text = await onetag.ask("Summarize this page in three short sentences, written to be read aloud. No lists, no markdown.", root, { maxTokens: 120 });
      speak((text.match(/[^.!?。！？]+[.!?。！？]*\s*/g) || [text]).map((s) => ({ text: s.trim(), el: null })));
    } catch (err) { now.classList.remove("sab-wait"); now.textContent = err.message; }
    sum.disabled = false;
  });
  addEventListener("pagehide", () => voice && voice.cancel());
</script>
