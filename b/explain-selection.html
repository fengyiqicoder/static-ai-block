<div id="explain-pop" hidden style="position:absolute;z-index:50;max-width:320px;padding:10px 12px;border-radius:10px;background:#111;color:#fff;font-size:14px;line-height:1.5;box-shadow:0 8px 24px rgba(0,0,0,.25)">

  <p id="explain-out" style="margin:8px 0 0;white-space:pre-wrap"></p>

<script type="module">
  const pop = document.getElementById("explain-pop");
  const btn = document.getElementById("explain-btn");
  const out = document.getElementById("explain-out");
  let selected = "";
  document.addEventListener("mouseup", () => {
    const sel = window.getSelection();
    const text = sel && sel.toString().trim();
    if (!text || text.length < 8 || pop.contains(sel.anchorNode)) return;
    const r = sel.getRangeAt(0).getBoundingClientRect();
    selected = text; out.textContent = "";
    pop.style.left = `${r.left + window.scrollX}px`;
    pop.style.top = `${r.bottom + window.scrollY + 8}px`;
    pop.hidden = false;
  });
  document.addEventListener("mousedown", (e) => { if (!pop.contains(e.target)) pop.hidden = true; });
  btn.addEventListener("click", async () => {
    out.textContent = "…";
    try {
      await onetag.ask("Explain this to a beginner in two sentences.", selected, {
        stream: (d) => { if (out.textContent === "…") out.textContent = ""; out.textContent += d; },
        maxTokens: 120,
      });
    } catch (err) { out.textContent = err.message; }
  });
</script>
