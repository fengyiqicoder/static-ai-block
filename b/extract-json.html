<script type="module">
  const btn = document.getElementById("ext-btn"), src = document.getElementById("ext-in"), out = document.getElementById("ext-out");
  btn.addEventListener("click", async () => {
    out.textContent = "…";
    try {
      const data = await onetag.ask(
        'Extract {"name": string, "company": string, "email": string, "quantity": number, "product": string, "deadline": "YYYY-MM-DD", "budget_eur": number}. Use null for missing fields.',
        src.value,
        { json: true }                       // json: true returns a parsed object
      );
      out.textContent = JSON.stringify(data, null, 2);
    } catch (err) { out.textContent = err.message; }
  });
</script>
