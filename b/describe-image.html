<script type="module">
  const file = document.getElementById("img-file"), img = document.getElementById("img-preview"), out = document.getElementById("img-out");
  async function describe(source) {
    out.textContent = "…";
    try {
      // An image File, Blob or <img> is sent as a picture (downscaled to 1,600 px by the SDK).
      const text = await onetag.ask("Write alt text for this image: one sentence, factual, no 'image of'.", source, { maxTokens: 80 });
      out.textContent = text; img.alt = text;
    } catch (err) { out.textContent = err.message; }
  }
  file.addEventListener("change", () => { const f = file.files[0]; if (f) { img.src = URL.createObjectURL(f); describe(f); } });
  document.getElementById("img-sample").addEventListener("click", () => describe(img));
</script>
