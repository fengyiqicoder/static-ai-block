<script type="module">
  const SYSTEM = "You are the help chat on this web page. Answer from the page content in one to three sentences. If the page does not cover it, say so.";
  const widget = document.getElementById("sab-launch"), panel = document.getElementById("launch-panel"), btn = document.getElementById("launch-btn");
  const log = document.getElementById("launch-log"), input = document.getElementById("launch-input");
  const page = document.querySelector("main") || document.body;
  const pageText = () => { widget.hidden = true; const text = page.innerText; widget.hidden = false; return text; };   // the page without this widget
  const history = [];

  const toggle = (open) => { panel.hidden = !open; btn.setAttribute("aria-expanded", String(open)); if (open) input.focus(); };
  btn.addEventListener("click", () => toggle(panel.hidden));
  document.getElementById("launch-close").addEventListener("click", () => toggle(false));
  document.addEventListener("keydown", (e) => { if (e.key === "Escape" && !panel.hidden) { toggle(false); btn.focus(); } });

  const say = (cls, text) => { const p = document.createElement("p"); p.className = cls; p.textContent = text; log.append(p); log.scrollTop = log.scrollHeight; return p; };
  document.getElementById("launch-form").addEventListener("submit", async (e) => {
    e.preventDefault();
    const q = input.value.trim(); if (!q) return;
    input.value = ""; say("launch-me", q);
    const out = say("launch-ai sab-wait", ""), text = pageText();
    input.focus();
    try {
      const a = await onetag.ask(q, text, { system: SYSTEM, history: history.slice(-6), maxTokens: 250,
        stream: (d) => { out.textContent += d; log.scrollTop = log.scrollHeight; } });
      history.push({ role: "user", content: q }, { role: "assistant", content: a });
    } catch (err) { out.textContent = err.message; }
  });
</script>
