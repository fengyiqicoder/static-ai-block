<pre><code>const debounce = (fn, ms) => { let t; return (...a) => { clearTimeout(t); t = setTimeout(() => fn(...a), ms); }; };</code></pre>
<pre><code>SELECT c.name, COUNT(o.id) FROM customers c LEFT JOIN orders o ON o.customer_id = c.id GROUP BY c.id HAVING COUNT(o.id) > 3;</code></pre>
<style>pre{position:relative;padding:12px 14px;background:#f1f5f9;border-radius:8px;overflow:auto}.code-explain{position:absolute;top:8px;right:8px;padding:3px 10px;border:1px solid #cbd5e1;border-radius:6px;background:#fff;font:13px system-ui;cursor:pointer}.code-note{margin:-4px 0 16px;padding:0 4px;font-size:14px;line-height:1.5;white-space:pre-wrap}</style>

<script type="module">
  for (const pre of document.querySelectorAll("pre")) {
    const btn = document.createElement("button"); btn.className = "code-explain"; btn.textContent = "Explain";
    const note = document.createElement("p"); note.className = "code-note";
    pre.append(btn); pre.after(note);
    btn.addEventListener("click", async () => {
      note.textContent = "";
      try {
        await onetag.ask("Explain what this code does in plain language, two or three sentences.", pre.innerText.replace("Explain", ""), {
          stream: (d) => { note.textContent += d; }, maxTokens: 150,
        });
      } catch (err) { note.textContent = err.message; }
    });
  }
</script>
