<script src="https://onetagai.com/v1.js" data-key="YOUR_ONETAG_KEY"></script>

<div style="max-width:560px">
  <textarea id="dr-in" rows="4" placeholder="Paste the email, review or support ticket here" style="width:100%;padding:10px 12px;border:1px solid #cbd5e1;border-radius:8px;font:inherit;box-sizing:border-box">Hi, I ordered the blue jacket (order 4471) two weeks ago and it still hasn't arrived. The tracking page hasn't updated since the 3rd. What's going on?</textarea>
  <div style="margin-top:8px;display:flex;gap:8px;align-items:center">
    <select id="dr-tone" style="padding:8px;border:1px solid #cbd5e1;border-radius:8px;font:inherit"><option>apologetic and helpful</option><option>brief and factual</option><option>warm</option></select>
    <button id="dr-btn" style="padding:8px 14px;border:0;border-radius:8px;background:#111;color:#fff;font:inherit;cursor:pointer">Draft reply</button>
  </div>
  <textarea id="dr-out" rows="6" style="width:100%;margin-top:8px;padding:10px 12px;border:1px solid #cbd5e1;border-radius:8px;font:inherit;box-sizing:border-box"></textarea>
</div>

<script type="module">
  const src = document.getElementById("dr-in"), tone = document.getElementById("dr-tone"), out = document.getElementById("dr-out");
  document.getElementById("dr-btn").addEventListener("click", async () => {
    out.value = "";
    try {
      await onetag.ask(`Write a reply to this message. Tone: ${tone.value}. Sign as "The team". Output only the reply.`, src.value, {
        stream: (d) => { out.value += d; }, maxTokens: 250,
      });
    } catch (err) { out.value = err.message; }
  });
</script>
