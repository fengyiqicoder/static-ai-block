<script type="module">
  const root = document.querySelector("article") || document.querySelector("main") || document.body;
  const url = location.href.startsWith("http") ? location.href : document.baseURI;
  const SKIP = "#sab-actions,script,style,noscript,nav,[hidden]";
  const BLOCKS = "h1,h2,h3,h4,h5,h6,p,pre,ul,ol,table,blockquote";

  // Text inside one block: links, bold, italic, inline code and images keep their Markdown form.
  const inline = (node) => [...node.childNodes].map((n) => {
    if (n.nodeType === 3) return n.textContent.replace(/\s+/g, " ");
    if (n.nodeType !== 1 || n.matches(SKIP)) return "";
    const tag = n.tagName, t = inline(n).trim();
    if (tag === "A") return t && n.href ? `[${t}](${n.href})` : t;
    if (tag === "STRONG" || tag === "B") return t && `**${t}**`;
    if (tag === "EM" || tag === "I") return t && `*${t}*`;
    if (tag === "CODE") return "`" + n.textContent + "`";
    if (tag === "IMG") return `![${n.alt}](${n.src})`;
    if (tag === "BR") return "\n";
    return inline(n);
  }).join("");

  function toMarkdown(node) {
    let md = "";
    for (const el of node.children) {
      if (el.matches(SKIP)) continue;
      const tag = el.tagName, text = () => inline(el).trim();
      if (/^H[1-6]$/.test(tag)) md += `${"#".repeat(+tag[1])} ${text()}\n\n`;
      else if (tag === "PRE") md += "```\n" + el.textContent.replace(/\n$/, "") + "\n```\n\n";
      else if (tag === "UL" || tag === "OL") md += [...el.children].map((li, i) => `${tag === "OL" ? `${i + 1}.` : "-"} ${inline(li).trim()}`).join("\n") + "\n\n";
      else if (tag === "TABLE") md += [...el.rows].map((row, i) => `| ${[...row.cells].map((c) => inline(c).trim()).join(" | ")} |` + (i ? "" : `\n|${" --- |".repeat(row.cells.length)}`)).join("\n") + "\n\n";
      else if (tag === "BLOCKQUOTE") md += (toMarkdown(el).trim() || text()).replace(/^/gm, "> ") + "\n\n";
      else if (el.querySelector(BLOCKS)) md += toMarkdown(el);            // a wrapper such as <div> or <section>: go inside
      else if (text()) md += `${text()}\n\n`;
    }
    return md;
  }
  const pageMarkdown = () => (root.querySelector("h1") || !document.title ? "" : `# ${document.title}\n\n`) + toMarkdown(root).trim() + `\n\nSource: ${url}\n`;

  const copy = document.getElementById("pa-copy"), view = document.getElementById("pa-view"), out = document.getElementById("pa-out");
  const showMarkdown = (open) => { out.textContent = open ? pageMarkdown() : ""; out.hidden = !open; view.setAttribute("aria-expanded", String(open)); };
  view.addEventListener("click", () => showMarkdown(out.hidden));
  copy.addEventListener("click", async () => {
    try { await navigator.clipboard.writeText(pageMarkdown()); copy.textContent = "Copied"; setTimeout(() => { copy.textContent = "Copy page as Markdown"; }, 1500); }
    catch { showMarkdown(true); }                                         // no clipboard access: show it so it can be selected
  });
  const prompt = encodeURIComponent(`Read ${url} so I can ask questions about it.`);
  document.getElementById("pa-gpt").href = `https://chatgpt.com/?hints=search&q=${prompt}`;
  document.getElementById("pa-claude").href = `https://claude.ai/new?q=${prompt}`;
</script>
