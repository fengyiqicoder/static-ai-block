<script type="module">
  const SYSTEM = "You are a helpful assistant on this website. Answer briefly, in Markdown.";   // who the assistant is
  const CONTEXT = null;      // optional knowledge: a string, or an element such as document.querySelector("main")
  const STORE = "sab-chat";  // conversations are kept in this browser only (localStorage)

  const $ = (id) => document.getElementById(id);
  const log = $("chat-log"), input = $("chat-input"), send = $("chat-send"), picker = $("chat-threads"), chips = $("chat-chips"), mic = $("chat-mic");
  const esc = (s) => s.replace(/[&<>"]/g, (c) => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;" }[c]));
  const THUMB = '<svg width="14" height="14" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round"><path d="M7 10v12M15 5.9 14 10h5.8a2 2 0 0 1 1.9 2.6l-2.3 8a2 2 0 0 1-1.9 1.4H4a2 2 0 0 1-2-2v-8a2 2 0 0 1 2-2h2.8a2 2 0 0 0 1.8-1.1L12 2a3.1 3.1 0 0 1 3 3.9z"/></svg>';

  // A small, safe Markdown renderer: everything is escaped first, then code, bold, italic, links, headings and lists are added back.
  const inline = (s) => s.split(/(`[^`\n]+`)/).map((p, i) => i % 2 ? `<code>${esc(p.slice(1, -1))}</code>` : esc(p)
    .replace(/\*\*([^*]+)\*\*/g, "<b>$1</b>").replace(/\*([^*\s][^*]*)\*/g, "<i>$1</i>")
    .replace(/\[([^\]]+)\]\((https?:\/\/[^\s)]+)\)/g, '<a href="$2" target="_blank" rel="noopener">$1</a>')).join("");
  function md(src) {
    return src.split("```").map((part, i) => {
      if (i % 2) return `<pre><button class="chat-copy" type="button">Copy</button><code>${esc(part.replace(/^[\w+-]*\n/, "").replace(/\n$/, ""))}</code></pre>`;
      let html = "", list = "";
      const endList = () => { if (list) html += `</${list}>`; list = ""; };
      for (const line of part.split("\n")) {
        const item = line.match(/^\s*(?:[-*]|(\d+)[.)])\s+(.*)/), head = line.match(/^#{1,6}\s+(.*)/);
        if (item) { const tag = item[1] ? "ol" : "ul"; if (list !== tag) { endList(); html += `<${tag}>`; list = tag; } html += `<li>${inline(item[2])}</li>`; continue; }
        endList();
        if (head) html += `<h4>${inline(head[1])}</h4>`; else if (line.trim()) html += `<p>${inline(line)}</p>`;
      }
      endList();
      return html;
    }).join("");
  }

  let threads = [];
  try { threads = JSON.parse(localStorage.getItem(STORE)) || []; } catch {}
  const newThread = () => { const t = { id: Date.now().toString(36), title: "New chat", messages: [] }; threads.unshift(t); return t; };
  const save = () => { try { localStorage.setItem(STORE, JSON.stringify(threads.filter((t) => t.messages.length).slice(0, 20))); } catch {} };
  let thread = threads[0] || newThread(), busy = null;   // busy: the AbortController of the answer being streamed

  function bubble(m, i) {
    const el = document.createElement("div");
    if (m.role === "user") { el.className = "chat-user"; el.textContent = m.content; return el; }
    el.className = "chat-ai"; el.dataset.i = i;
    el.innerHTML = `<div class="chat-md">${md(m.content)}</div><div class="chat-acts"><button type="button" data-act="copy">Copy</button>${i === thread.messages.length - 1 ? '<button type="button" data-act="retry">Retry</button>' : ""}
      <button type="button" data-act="up" aria-label="Good answer" aria-pressed="${m.vote === "up"}">${THUMB}</button><button type="button" data-act="down" aria-label="Bad answer" aria-pressed="${m.vote === "down"}" style="transform:rotate(180deg)">${THUMB}</button></div>`;
    return el;
  }
  function draw() {
    picker.innerHTML = threads.map((t) => `<option value="${t.id}">${esc(t.title)}</option>`).join(""); picker.value = thread.id;
    log.replaceChildren(...thread.messages.map(bubble));
    chips.hidden = thread.messages.length > 0; log.scrollTop = log.scrollHeight;
  }

  async function answer() {                              // the last message is the user's; stream the reply under it
    const t = thread, q = t.messages.at(-1).content, history = t.messages.slice(0, -1).slice(-6).map(({ role, content }) => ({ role, content }));
    const m = { role: "assistant", content: "" }; t.messages.push(m); draw();
    const out = log.lastElementChild.querySelector(".chat-md"); out.classList.add("sab-wait");
    busy = new AbortController(); send.textContent = "Stop";
    try {
      await onetag.ask(q, CONTEXT, { system: SYSTEM, history, maxTokens: 400, signal: busy.signal,
        stream: (d) => { m.content += d; out.innerHTML = md(m.content); log.scrollTop = log.scrollHeight; } });
    } catch (err) {
      if (err.code !== "aborted") {                      // a failed turn is taken back so the question can be sent again
        t.messages.splice(-2); busy = null; send.textContent = "Send"; draw(); input.value = q;
        const p = document.createElement("p"); p.className = "chat-err"; p.textContent = err.message; log.append(p);
        return;
      }
      if (!m.content) m.content = "*Stopped.*";
    }
    if (t.title === "New chat") t.title = q.slice(0, 40);
    busy = null; send.textContent = "Send"; save(); draw();
  }
  const ask = (q) => { thread.messages.push({ role: "user", content: q }); answer(); };

  $("chat-form").addEventListener("submit", (e) => {
    e.preventDefault();
    if (busy) return busy.abort();                       // while an answer streams, the button is Stop
    const q = input.value.trim(); if (!q) return;
    input.value = ""; ask(q);
  });
  for (const chip of chips.children) chip.addEventListener("click", () => { if (!busy) ask(chip.textContent); });
  picker.addEventListener("change", () => { if (busy) return draw(); thread = threads.find((t) => t.id === picker.value); draw(); });
  $("chat-new").addEventListener("click", () => { if (busy || !thread.messages.length) return; thread = newThread(); draw(); input.focus(); });
  $("chat-del").addEventListener("click", () => { if (busy) return; threads = threads.filter((t) => t !== thread); thread = threads[0] || newThread(); save(); draw(); });

  log.addEventListener("click", async (e) => {
    const btn = e.target.closest("button"); if (!btn) return;
    const flash = () => { const t = btn.textContent; btn.textContent = "Copied"; setTimeout(() => { btn.textContent = t; }, 1500); };
    if (btn.className === "chat-copy") { await navigator.clipboard.writeText(btn.nextElementSibling.textContent).then(flash, () => {}); return; }
    const i = +btn.closest(".chat-ai").dataset.i, m = thread.messages[i], act = btn.dataset.act;
    if (act === "copy") await navigator.clipboard.writeText(m.content).then(flash, () => {});
    if (act === "retry" && !busy) { thread.messages.splice(i); answer(); }
    if (act === "up" || act === "down") {
      m.vote = m.vote === act ? null : act; save();
      for (const b of btn.parentNode.querySelectorAll("[aria-pressed]")) b.setAttribute("aria-pressed", String(b.dataset.act === m.vote));
      // Listen for this event to send votes to your analytics: document.addEventListener("sab-feedback", (e) => console.log(e.detail))
      if (m.vote) btn.dispatchEvent(new CustomEvent("sab-feedback", { bubbles: true, detail: { vote: m.vote, question: thread.messages[i - 1].content, answer: m.content } }));
    }
  });

  const Speech = window.SpeechRecognition || window.webkitSpeechRecognition;   // voice input, where the browser has it; costs no tokens
  if (Speech) {
    const rec = new Speech(); let on = false, typed = "";
    rec.lang = document.documentElement.lang || navigator.language; rec.interimResults = true; mic.hidden = false;
    rec.onresult = (e) => { input.value = typed + [...e.results].map((r) => r[0].transcript).join(""); };
    rec.onend = () => { on = false; mic.setAttribute("aria-pressed", "false"); };
    mic.addEventListener("click", () => { if (on) return rec.stop(); typed = input.value ? input.value + " " : ""; on = true; mic.setAttribute("aria-pressed", "true"); rec.start(); });
  }
  draw();
</script>
