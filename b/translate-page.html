<script type="module">
  const btn = document.getElementById("tr-btn"), lang = document.getElementById("tr-lang"), note = document.getElementById("tr-note");
  const target = document.querySelector("article") || document.querySelector("main") || document.body;
  const browserLang = new Intl.DisplayNames(["en"], { type: "language" }).of(navigator.language.split("-")[0]);
  for (const o of lang.options) if (o.text === browserLang) lang.value = o.text;
  let original = null;
  btn.addEventListener("click", async () => {
    if (original) { target.innerHTML = original; original = null; btn.textContent = "Translate"; note.textContent = ""; return; }
    btn.disabled = true; note.textContent = "Translating…";
    try {
      // Line per block keeps the layout and keeps long pages under the 32,000-character limit.
      const blocks = [...target.querySelectorAll("h1,h2,h3,p,li")].filter((el) => el.innerText.trim() && !el.closest("#sab-translate"));
      const text = await onetag.ask(`Translate each line to ${lang.value}. Keep the line order and count exactly; one output line per input line; no numbering.`,
        blocks.map((el) => el.innerText.replace(/\n/g, " ")).join("\n"), { maxTokens: 1000 });
      original = target.innerHTML;
      const lines = text.split("\n");
      blocks.forEach((el, i) => { if (lines[i]) el.innerText = lines[i]; });
      btn.textContent = "Show original"; note.textContent = "";
    } catch (err) { note.textContent = err.message; }
    btn.disabled = false;
  });
</script>
