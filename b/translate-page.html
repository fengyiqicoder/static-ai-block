<script type="module">
  const btn = document.getElementById("tr-btn"), lang = document.getElementById("tr-lang"), both = document.getElementById("tr-both"), note = document.getElementById("tr-note");
  const target = document.querySelector("article") || document.querySelector("main") || document.body;
  // Start on the visitor's language, unless the page is already written in it.
  const codes = [...lang.options].map((o) => o.value), pageLang = (document.documentElement.lang || "en").split("-")[0];
  lang.value = [navigator.language.split("-")[0], ...codes].find((c) => c !== pageLang && codes.includes(c));
  let saved = null;                                     // [element, its original HTML] for every translated block
  btn.addEventListener("click", async () => {
    if (saved) { for (const [el, html] of saved) el.innerHTML = html; saved = null; btn.textContent = "Translate"; note.textContent = ""; return; }
    // One numbered line per block keeps the layout, and a skipped or merged line cannot shift the rest.
    const blocks = [...target.querySelectorAll("h1,h2,h3,h4,p,li,td,th,blockquote")].filter((el) => el.innerText.trim() && !el.querySelector("p,li") && !el.closest("#sab-translate"));
    btn.disabled = true; note.textContent = "Translating…";
    let got = "";
    try {
      const to = new Intl.DisplayNames(["en"], { type: "language" }).of(lang.value);        // "zh" becomes "Chinese"
      const text = await onetag.ask(`Translate each numbered line to ${to}. Reply with every line in the same form, [number] translation, and nothing else.`,
        blocks.map((el, i) => `[${i + 1}] ${el.innerText.replace(/\s+/g, " ")}`).join("\n"), {
          maxTokens: 4000,
          stream: (d) => { got += d; note.textContent = `Translating… ${Math.min((got.match(/^\[\d+\]/gm) || []).length, blocks.length)} of ${blocks.length}`; },
        });
      const lines = [];
      for (const [, n, line] of text.matchAll(/^\[(\d+)\]\s*(.+)$/gm)) lines[n - 1] = line.trim();
      if (!lines.length) throw new Error("The translation came back empty. Try again.");
      saved = blocks.map((el) => [el, el.innerHTML]);
      blocks.forEach((el, i) => {
        if (!lines[i]) return;
        if (!both.checked) { el.innerText = lines[i]; return; }
        const added = document.createElement("span");    // bilingual: the translation sits under the original
        added.style.cssText = "display:block;opacity:.7"; added.textContent = lines[i]; el.append(added);
      });
      btn.textContent = "Show original"; note.textContent = "";
    } catch (err) { note.textContent = err.message; }
    btn.disabled = false;
  });
</script>
