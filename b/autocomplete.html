<script type="module">
  const WAIT = 700;    // ms of pause before asking
  const MIN = 12;      // characters needed before the first suggestion
  const LIMIT = 20;    // suggestions per visit; each one costs about 200 tokens
  const box = document.getElementById("ac-text"), typed = document.getElementById("ac-typed"), ghost = document.getElementById("ac-hint");
  const accept = document.getElementById("ac-accept"), note = document.getElementById("ac-note");
  let hint = "", timer, request, used = 0;

  // The suggestion is drawn in a layer under the textarea: the typed text is invisible there, the suggestion is grey.
  const show = (s) => { hint = s; typed.textContent = box.value; ghost.textContent = s + "\u200b"; accept.hidden = !s; };

  async function suggest() {
    const text = box.value;
    request = new AbortController(); used++;
    try {
      let s = (await onetag.ask("Continue this text with the next 3 to 10 words. Output only the continuation: no quotes, no repetition of the text.",
        text.slice(-600), { maxTokens: 24, signal: request.signal })).trim().replace(/^["“]|["”]$/g, "");
      // Drop any words the model repeated from the end of the text.
      const tail = text.trimEnd().toLowerCase(), words = s.split(/\s+/);
      for (let n = words.length; n > 0; n--) {
        const lead = words.slice(0, n).join(" ").toLowerCase();
        if (tail.endsWith(lead) && !/\S/.test(tail.charAt(tail.length - lead.length - 1))) { s = words.slice(n).join(" "); break; }
      }
      if (box.value !== text || !s) return;                                   // the text moved on while we waited
      if (!/\s$/.test(text) && !/^[.,;:!?)']/.test(s)) s = " " + s;
      show(s);
    } catch (err) { if (err.code !== "aborted") note.textContent = err.message; }
  }
  function plan() {
    show(""); clearTimeout(timer); if (request) request.abort();
    if (box.value.trim().length < MIN || box.selectionStart !== box.value.length) return;   // only at the end of the text
    if (used >= LIMIT) { note.textContent = "Suggestion limit for this visit reached."; return; }
    timer = setTimeout(suggest, WAIT);
  }
  const take = () => { if (!hint) return; box.value += hint; box.focus(); plan(); };

  box.addEventListener("input", plan);
  box.addEventListener("keydown", (e) => {
    if (e.key === "Tab" && hint) { e.preventDefault(); take(); }
    else if (e.key === "Escape" && hint) show("");
  });
  accept.addEventListener("click", take);
  show("");
</script>
